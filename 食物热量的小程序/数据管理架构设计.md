#### 2.2 缓存更新策略
```javascript
// 缓存更新策略
class CacheManager {
  // 写入时更新缓存
  async updateCache(key, data, expiration) {
    await redis.setex(key, expiration, JSON.stringify(data));
  }
  
  // 读取时检查缓存
  async getFromCache(key) {
    const cached = await redis.get(key);
    return cached ? JSON.parse(cached) : null;
  }
  
  // 删除相关缓存
  async invalidateUserCache(userId) {
    const pattern = `user:*:${userId}*`;
    const keys = await redis.keys(pattern);
    if (keys.length > 0) {
      await redis.del(...keys);
    }
  }
}
```

### 3. 数据同步机制

#### 3.1 实时同步策略
```javascript
// WebSocket实时同步
class DataSyncManager {
  constructor() {
    this.wsConnections = new Map();
    this.syncQueue = new Queue();
  }
  
  // 用户连接时建立同步通道
  async establishSync(userId, wsConnection) {
    this.wsConnections.set(userId, wsConnection);
    await this.syncPendingData(userId);
  }
  
  // 数据变更时推送更新
  async pushUpdate(userId, dataType, data) {
    const connection = this.wsConnections.get(userId);
    if (connection) {
      connection.send(JSON.stringify({
        type: 'data_update',
        dataType,
        data,
        timestamp: Date.now()
      }));
    } else {
      // 用户离线，加入同步队列
      await this.syncQueue.add(userId, dataType, data);
    }
  }
}
```

#### 3.2 离线数据同步
```javascript
// 离线数据管理
class OfflineDataManager {
  // 本地存储离线数据
  async storeOfflineData(userId, data) {
    const offlineKey = `offline:${userId}`;
    const existingData = await this.getOfflineData(userId);
    const newData = [...existingData, { ...data, timestamp: Date.now() }];
    await localStorage.setItem(offlineKey, JSON.stringify(newData));
  }
  
  // 网络恢复时同步数据
  async syncOfflineData(userId) {
    const offlineData = await this.getOfflineData(userId);
    for (const data of offlineData) {
      try {
        await this.syncToServer(data);
        await this.removeOfflineData(userId, data.id);
      } catch (error) {
        console.error('同步失败:', error);
      }
    }
  }
}
```

```javascript
// 数据加密工具
class DataEncryption {
  // 敏感数据加密
  encryptSensitiveData(data, key) {
    const cipher = crypto.createCipher('aes-256-cbc', key);
    let encrypted = cipher.update(JSON.stringify(data), 'utf8', 'hex');
    encrypted += cipher.final('hex');
    return encrypted;
  }
  
  // 数据解密
  decryptSensitiveData(encryptedData, key) {
    const decipher = crypto.createDecipher('aes-256-cbc', key);
    let decrypted = decipher.update(encryptedData, 'hex', 'utf8');
    decrypted += decipher.final('utf8');
    return JSON.parse(decrypted);
  }
}
```

```javascript
// 数据质量检查
class DataQualityMonitor {
  async checkDataQuality() {
    const checks = [
      this.checkDuplicateRecords(),
      this.checkInvalidNutritionData(),
      this.checkOrphanedRecords(),
      this.checkDataConsistency()
    ];
    
    const results = await Promise.all(checks);
    return this.generateQualityReport(results);
  }
  
  // 数据一致性检查
  async checkDataConsistency() {
    const inconsistencies = [];
    
    // 检查饮食记录与详情表的一致性
    const inconsistentRecords = await db.query(`
      SELECT dr.record_id, dr.total_calories, SUM(dri.calories) as calculated_calories
      FROM diet_records dr
      LEFT JOIN diet_record_items dri ON dr.record_id = dri.record_id
      GROUP BY dr.record_id
      HAVING ABS(dr.total_calories - calculated_calories) > 0.01
    `);
    
    if (inconsistentRecords.length > 0) {
      inconsistencies.push({
        type: 'calorie_mismatch',
        count: inconsistentRecords.length,
        records: inconsistentRecords
      });
    }
    
    return inconsistencies;
  }
}
```
